\newcommand{\EqKinematics}{\begin{equation}\label{eq:kinematics}
\nu=\Rot\T\bm r_s,\quad \kappa=(\Rot\T\Rot_s)^\vee,
\quad \Omega=(\Rot\T\Rot_t)^\vee,\quad \Rot\in\SO(3).\tag{P1}
\end{equation}}
\newcommand{\EqRodBalance}{\begin{equation}\label{eq:rod-balance}\begin{aligned}
\mu_r\bm r_{tt}&=\bm n_s+\bm f_m+\bm f_f+\bm f_c+\bm f_g,\\
\Inertia\dot\Omega+\Omega\times(\Inertia\Omega)
 &=\Rot\T(\bm m_s+\bm r_s\times\bm n+\ell_m+\ell_f+\ell_c).
\end{aligned}\tag{P2}\end{equation}}
\newcommand{\EqConstitutive}{\begin{equation}\label{eq:constitutive}\begin{aligned}
W&=\tfrac12(\nu-\nu_0)\T C_\nu(\nu-\nu_0)
 +\tfrac12(\kappa-\kappa_0)\T C_\kappa(\kappa-\kappa_0),\\
\bm n&=\Rot W_\nu,\qquad \bm m=\Rot W_\kappa.
\end{aligned}\tag{P3}\end{equation}}
\newcommand{\EqMagneticEnergy}{\begin{equation}\label{eq:magnetic-energy}\begin{aligned}
V_m&=-\int_0^L\mu(s)\T B(\bm r(s),u,t)\,\dd s,\qquad \mu=\Rot\mu_b,\\
\bm f_m&=(\nabla B)\T\mu,\qquad \ell_m=\mu\times B.
\end{aligned}\tag{P4}\end{equation}}
\newcommand{\EqMagneticPort}{\begin{equation}\label{eq:magnetic-port}\begin{aligned}
B(x)&=b+G(x-x_0),\quad G=G\T,\quad \tr G=0,\\
V_m(q,u)&=-\phi(q)\T u,\quad Q_m=\Gmag(q)u,
\quad \Gmag=(D_q\phi)\T,\quad P_m=\dot q\T\Gmag u.
\end{aligned}\tag{P5}\end{equation}}
\newcommand{\EqFluid}{\begin{equation}\label{eq:fluid}\begin{aligned}
\rho_f\big[\partial_t^{\mathrm{ALE}}u_f+((u_f-w_{\mathrm{ALE}})\cdot\nabla)u_f\big]
 &=\nabla\cdot\sigma_f+\rho_f g_{\mathrm{eff}},\\
\nabla\cdot u_f&=0,\qquad \sigma_f=-p_fI+2\eta D(u_f).
\end{aligned}\tag{P6}\end{equation}}
\newcommand{\EqContact}{\begin{equation}\label{eq:contact}\begin{aligned}
\delta&=[-g]_+,\qquad V_c=\int_0^L\frac{k_c}{r_c+1}\delta^{r_c+1}\,\dd s,
\quad r_c=\tfrac32,\\
\bm f_{c,e}&=-k_c\delta^{r_c}\bm n_c,\qquad
\bm f_\tau=-\mu_f\lambda_n\tanh\!\left(\frac{|v_\tau|}{v_\epsilon}\right)\frac{v_\tau}{|v_\tau|}.
\end{aligned}\tag{P7}\end{equation}}
\newcommand{\EqMass}{\begin{equation}\label{eq:mass}
M(q)=\int_0^L\left[\mu_rJ_x\T J_x+J_\Omega\T\Inertia J_\Omega\right]\dd s+M_{\mathrm{tip}}(q),
\quad T=\tfrac12\dot q\T M(q)\dot q.\tag{P8}\end{equation}}
\newcommand{\EqHamiltonian}{\begin{equation}\label{eq:hamiltonian}\begin{aligned}
p&=M(q)\dot q,\qquad H(q,p)=\tfrac12p\T M(q)^{-1}p+V(q),\\
\dot q&=H_p=v,\qquad
\dot p_i=\tfrac12v\T(\partial_i M)v-\partial_iV+Q_{\mathrm{nc},i}.
\end{aligned}\tag{P9}\end{equation}}
\newcommand{\EqPower}{\begin{equation}\label{eq:power}
\dot E_{\mathrm{mech}}=P_m+P_f+P_{c,d}+P_g+P_b-P_{\mathrm{loss}}.\tag{P10}\end{equation}}
\newcommand{\EqBirkhoff}{\begin{equation}\label{eq:birkhoff}\begin{aligned}
\theta&=a_i\dd z_i-\Birk\,\dd t,\qquad
\Kform_{ij}=\partial_i a_j-\partial_j a_i,\\
\Kform f&=\nabla\Birk+\partial_t\Pfaff,\qquad \det\Kform\ne0.
\end{aligned}\tag{A1}\end{equation}}
\newcommand{\EqCompatibility}{\begin{equation}\label{eq:compatibility}\begin{aligned}
\dd_z\Omega_t&=0,\qquad \Omega_t=\tfrac12\Kform_{ij}\dd z_i\wedge\dd z_j,\\
\partial_t\Kform+(f\cdot\nabla)\Kform+(Df)\T\Kform+\Kform Df&=0.
\end{aligned}\tag{A2}\end{equation}}
\newcommand{\EqTransport}{\begin{equation}\label{eq:transport}\begin{aligned}
\partial_t\Pfaff+(D\Pfaff)f+(Df)\T\Pfaff&=0,\qquad \Birk=\Pfaff\T f,\\
D\Phi_{t,t_0}\T\Kform(\Phi_{t,t_0}(z),t)D\Phi_{t,t_0}&=\Kform(z,t_0).
\end{aligned}\tag{A3}\end{equation}}
\newcommand{\EqEndpoint}{\begin{equation}\label{eq:endpoint}\begin{aligned}
D\alpha\T\mathbb J D\alpha&=\diag(\Kform(z_+,t_+),-\Kform(z_-,t_-)),\\
N&=(A_\alpha F+B_\alpha)(C_\alpha F+D_\alpha)^{-1}=N\T,

\end{aligned}\tag{A4}\end{equation}}
\newcommand{\EqInitialGenerating}{\begin{equation}\label{eq:initial-generating}\begin{aligned}

w&=\alpha_2(z,z;t_0,t_0),\qquad f_0(w)=\alpha_1(z,z;t_0,t_0).

\newcommand{\EqGeneratingTime}{\begin{equation}\label{eq:generating-time}

\newcommand{\EqMidpoint}{\begin{equation}\label{eq:midpoint}\begin{aligned}
q_+-q_-&=hM(\overline q,\overline t)^{-1}\overline p,\\
p_{+,i}-p_{-,i}&=h\left[\tfrac12\overline v\T(\partial_iM)\overline v-\partial_i V+Q_{\mathrm{nc},i}\right]_{\overline z,\overline t}.

\newcommand{\EqOriginalThirtySix}{\begin{equation}\label{eq:original36}\begin{aligned}

\newcommand{\EqAllocation}{\begin{equation}\label{eq:allocation}\begin{aligned}
